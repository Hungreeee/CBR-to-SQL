give me the number of patients who have been diagnosed with tobacco use disorder.
----------
PREDS:
SELECT COUNT ( DISTINCT DEMOGRAPHIC."SUBJECT_ID" ) 
FROM DEMOGRAPHIC 
INNER JOIN DIAGNOSES on DEMOGRAPHIC.HADM_ID = DIAGNOSES.HADM_ID 
WHERE DIAGNOSES."LONG_TITLE" = 'Tobacco use disorder'
{'sel': 'count', 'agg': ['demographic."subject_id"'], 'tab': ['demographic', 'diagnoses'], 'cond': [['diagnoses."long_title"', '=', 'tobacco use disorder']]}
